NAME:
   ant beta:sessions create - Create Session

USAGE:
   ant beta:sessions create

OPTIONS:
   --agent value                                                      Agent identifier. Accepts the `agent` ID string, which pins the latest version for the session, or an `agent` object with both id and version specified.
     --agent.id string                                                The `agent` ID.
     --agent.type string                                              Allowed values: "agent", "agent_with_overrides".
     --agent.mcp-servers '{key: value}'                               Replacement MCP server list. Full replacement: the provided array becomes the MCP servers. Send an empty array to clear; omit to preserve the agent's servers.
     --agent.model value                                              Replacement model. Accepts the model string, e.g. `claude-opus-5`, or a `model_config` object. Omit to use the agent's model.
     --agent.skills '{key: value}'                                    Replacement skill list. Full replacement: the provided array becomes the skills. Send an empty array to clear; omit to preserve the agent's skills.
     --agent.system string                                            Replacement system prompt. Up to 100,000 characters. Set to null to clear the agent's system prompt; omit to preserve it.
     --agent.tools '{key: value}'                                     Replacement tool list. Full replacement: the provided array becomes the tool configuration. Send an empty array to clear; omit to preserve the agent's tools.
     --agent.version int                                              The specific `agent` version to use. Omit to use the latest version. Must be at least 1 if specified.
   --environment-id string                                            ID of the `environment` defining the container configuration for this session.
   --budget '{key: value}'                                            A hard spend ceiling. The session stops issuing new model requests once the tracked list cost reaches `max_list_cost`.
     --budget.max-list-cost '{key: value}'                            A monetary amount in a specific currency.
     --budget.type string                                             Allowed values: "limit".
   --initial-event '{key: value}' [ --initial-event '{key: value}' ]  Initial events to send to the `session` at creation, processed in order. Supports `user.message` and `user.define_outcome` events. Maximum 50 events.
     --initial-event.type string                                      Allowed values: "user.message", "user.define_outcome".
     --initial-event.content '{key: value}'                           Array of content blocks for the user message.
     --initial-event.description string                               What the agent should produce. This is the task specification.
     --initial-event.max-iterations int                               Eval→revision cycles before giving up. Default 3, max 20.
     --initial-event.rubric '{key: value}'                            How to grade the outcome. Text or file reference.
   --metadata '{key: value}'                                          Arbitrary key-value metadata attached to the session. Maximum 16 pairs, keys up to 64 chars, values up to 512 chars.
   --resource '{key: value}' [ --resource '{key: value}' ]            Resources (e.g. repositories, files) to mount into the session's container.
     --resource.type string                                           Allowed values: "github_repository", "file", "memory_store".
     --resource.access string                                         Access mode for the mounted store. Defaults to read_write. read_only mounts the store as a read-only filesystem.
     --resource.authorization-token string                            GitHub authorization token used to clone the repository. Required for private repositories; optional for public ones.
     --resource.checkout '{key: value}'                               Branch or commit to check out. Defaults to the repository's default branch.
     --resource.file-id string                                        ID of a previously uploaded file.
     --resource.instructions string                                   Per-attachment guidance for the agent on how to use this store. Rendered into the memory section of the system prompt. Max 4096 chars.
     --resource.memory-store-id string                                The memory store ID (memstore_...). Must belong to the caller's organization and workspace.
     --resource.mount-path string                                     Mount path in the container. Defaults to `/workspace/<repo-name>`.
     --resource.url string                                            Github URL of the repository
   --title string                                                     Human-readable session title.
   --vault-id string [ --vault-id string ]                            Vault IDs for stored credentials the agent can use during the session.
   --beta string [ --beta string ]                                    Optional header to specify the beta version(s) you want to use.
   --workspace-id string                                              Optional header to select the Workspace for this request. The value is a Workspace ID (for example, `wrkspc_011CZkZaBF1tNoB5wlCeusgy`).

      Only needed for credentials that can act on more than one Workspace. A credential that belongs to a specific Workspace may omit it; if sent, it must match that Workspace.
   --help, -h  show help
